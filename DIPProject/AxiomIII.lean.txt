import Mathlib.CategoryTheory.Adjunction.Basic
import Mathlib.CategoryTheory.Types

open CategoryTheory

namespace DIP.AxiomIII

inductive OnePoint : Type where
  | pt

open OnePoint

instance instCat : Category OnePoint where
  Hom _ _ := PUnit
  id _ := PUnit.unit
  comp _ _ := PUnit.unit
  id_comp _ := rfl
  comp_id _ := rfl
  assoc _ _ _ := rfl

instance instSub (a b : OnePoint) : Subsingleton (CategoryStruct.Hom a b) :=
  Subsingleton.mk (fun _ _ => rfl)

def constToType (A : Type) : Functor OnePoint Type where
  obj _ := A
  map _ := id
  map_id _ := rfl
  map_comp _ _ := rfl

theorem no_adjoint_of_nontrivial_target {A : Type} [Nontrivial A] :
    Not (Exists (fun (i_star : Functor Type OnePoint) =>
      Nonempty (Adjunction (constToType A) i_star))) := by
  intro h
  let i_star := h.choose
  have h_adj : Nonempty (Adjunction (constToType A) i_star) := h.choose_spec
  let adj := Classical.choice h_adj
  have h_equiv : Equiv (A -> Bool)
      (CategoryStruct.Hom OnePoint.pt (i_star.obj Bool)) :=
    adj.homEquiv OnePoint.pt Bool
  have h_sub : Subsingleton (A -> Bool) :=
    Subsingleton.mk (fun f g => h_equiv.injective (Subsingleton.elim _ _))
  have h_eq : (fun _ : A => true) = (fun _ : A => false) :=
    h_sub.elim (fun _ => true) (fun _ => false)
  have h_ne : Exists (fun a1 => Exists (fun a2 : A => Not (a1 = a2))) :=
    exists_pair_ne A
  let a1 := h_ne.choose
  have h_tf : (true : Bool) = false := congrFun h_eq a1
  cases h_tf

end DIP.AxiomIII
