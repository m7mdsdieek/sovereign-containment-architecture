'inductive OnePoint : Type where'

'  comp _ _ := PUnit.unit'

'  id_comp _ := rfl'
'  comp_id _ := rfl'
'  assoc _ _ _ := rfl'

'instance instSub (a b : OnePoint) : Subsingleton (Quiver.Hom a b) :='
'  Subsingleton.intro (fun _ _ => rfl)'

'  obj _ := A'
'  map _ := id'
'  map_id _ := rfl'
'  map_comp _ _ := rfl'

'theorem no_adjoint_of_nonempty_target {A : Type} [Nonempty A] :'
'    Not (Exists (fun (i_star : Functor Type OnePoint) =>'
'      Nonempty (Adjunction (constToType A) i_star))) := by'
'  intro h'
'  let i_star := h.choose'
'  have h_adj : Nonempty (Adjunction (constToType A) i_star) := h.choose_spec'
'  let adj := Classical.choice h_adj'
'  have h_equiv : Equiv (A -> Bool) (Quiver.Hom OnePoint.pt (i_star.obj Bool)) :='
'    adj.homEquiv OnePoint.pt Bool'
'  haveI : Subsingleton (Quiver.Hom OnePoint.pt (i_star.obj Bool)) := inferInstance'
'  have h_sub : Subsingleton (A -> Bool) :='
'    Subsingleton.intro (fun f g => h_equiv.injective (Subsingleton.elim _ _))'
'  have h_fg : (fun _ : A => true) = (fun _ : A => false) := h_sub.elim _ _'
'  have hne : Nonempty A := inferInstance'

'  cases htf'
