$c = @'
import Mathlib.CategoryTheory.Functor.FullyFaithful
import Mathlib.CategoryTheory.Category.Basic

open CategoryTheory

universe v1 v2 u1 u2

namespace DIP.AxiomI

variable {O : Type u1} [Category.{v1} O]
variable {C : Type u2} [Category.{v2} C]
variable (T : Functor O C)

theorem tool_faithful_injective [Faithful T] {A B : O} :
    Function.Injective
      (T.map : CategoryStruct.Hom A B -> CategoryStruct.Hom (T.obj A) (T.obj B)) :=
  T.map_injective

theorem no_agency_merging [Faithful T] {A B : O} {f g : CategoryStruct.Hom A B}
    (h : T.map f = T.map g) : f = g :=
  T.map_injective h

theorem empty_propagation [Full T] {A B : O}
    (h : CategoryStruct.Hom A B -> False) :
    CategoryStruct.Hom (T.obj A) (T.obj B) -> False := by
  intro f
  exact h (T.map_surjective f).choose

theorem hom_equiv [Full T] [Faithful T] {A B : O} :
    Nonempty (Equiv (CategoryStruct.Hom A B) (CategoryStruct.Hom (T.obj A) (T.obj B))) :=
  Nonempty.intro (Equiv.ofBijective T.map
    (And.intro T.map_injective T.map_surjective))

theorem no_autonomous_endomorphism [Full T] [Faithful T] {A : O}
    (h : forall (f : CategoryStruct.Hom A A), f = CategoryStruct.id A)
    (g : CategoryStruct.Hom (T.obj A) (T.obj A)) :
    g = CategoryStruct.id (T.obj A) := by
  let f := (T.map_surjective g).choose
  have hf : T.map f = g := (T.map_surjective g).choose_spec
  rw [<- hf, h f, T.map_id]

end DIP.AxiomI
