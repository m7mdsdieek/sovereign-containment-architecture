$lines = @(
'import Mathlib.CategoryTheory.Category.Basic'
'import Mathlib.Algebra.Order.Monoid.Defs'
''
'open CategoryTheory'
''
'namespace DIP.AxiomIV'
''
'universe v u'
''
'variable {C : Type u} [Category.{v} C]'
'variable {E : Type} [OrderedAddCommMonoid E]'
''
'inductive Path : C -> C -> Type (max u v) where'
'  | nil {X : C} : Path X X'
'  | cons {X Y Z : C} (f : Quiver.Hom X Y) (p : Path Y Z) : Path X Z'
''
'namespace Path'
''
'def gradeOf (grade : {X Y : C} -> Quiver.Hom X Y -> E) :'
'    {X Y : C} -> Path X Y -> E'
'  | _, _, .nil => 0'
'  | _, _, .cons f p => grade f + gradeOf grade p'
''
'def append : {X Y Z : C} -> Path X Y -> Path Y Z -> Path X Z'
'  | _, _, _, .nil, q => q'
'  | _, _, _, .cons f p, q => .cons f (append p q)'
''
'theorem gradeOf_append (grade : {X Y : C} -> Quiver.Hom X Y -> E)'
'    {X Y Z : C} (p : Path X Y) (q : Path Y Z) :'
'    gradeOf grade (append p q) = gradeOf grade p + gradeOf grade q := by'
'  induction p with'
'  | nil => simp [append, gradeOf]'
'  | cons f p ih =>'
'    simp only [append, gradeOf, ih]'
'    rw [add_assoc]'
''
'theorem gradeOf_nonneg (grade : {X Y : C} -> Quiver.Hom X Y -> E)'
'    (grade_nonneg : forall {X Y : C} (f : Quiver.Hom X Y), LE.le (0 : E) (grade f))'
'    {X Y : C} (p : Path X Y) : LE.le (0 : E) (gradeOf grade p) := by'
'  induction p with'
'  | nil => exact le_refl 0'
'  | cons f p ih => exact add_nonneg (grade_nonneg f) ih'
''
'def withinBudget (grade : {X Y : C} -> Quiver.Hom X Y -> E)'
'    (colim_sigma : E) {X Y : C} (p : Path X Y) : Prop :='
'  LE.le (gradeOf grade p) colim_sigma'
''
'theorem stasis (grade : {X Y : C} -> Quiver.Hom X Y -> E)'
'    (grade_nonneg : forall {X Y : C} (f : Quiver.Hom X Y), LE.le (0 : E) (grade f))'
'    (colim_sigma : E) {X Y : C} (p : Path X Y)'
'    (h : Not (withinBudget grade colim_sigma p))'
'    {Z : C} (q : Path Y Z) :'
'    Not (withinBudget grade colim_sigma (append p q)) := by'
'  intro h_ext'
'  apply h'
'  unfold withinBudget at *'
'  rw [gradeOf_append grade p q] at h_ext'
'  have hq : LE.le (0 : E) (gradeOf grade q) := gradeOf_nonneg grade grade_nonneg q'
'  have h1 : LE.le (gradeOf grade p) (gradeOf grade p + gradeOf grade q) := by'
'    have step : LE.le (gradeOf grade p + 0) (gradeOf grade p + gradeOf grade q) :='
'      add_le_add_left hq _'
'    simpa using step'
'  exact le_trans h1 h_ext'
''
'end Path'
'end DIP.AxiomIV'
)
