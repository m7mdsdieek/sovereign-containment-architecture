$lines = @(
'import Mathlib.CategoryTheory.Functor.FullyFaithful'
'import Mathlib.CategoryTheory.Category.Basic'
''
'open CategoryTheory'
''
'universe v1 v2 u1 u2'
''
'namespace DIP.AxiomI'
''
'variable {O : Type u1} [Category.{v1} O]'
'variable {C : Type u2} [Category.{v2} C]'
'variable (T : Functor O C)'
''
'theorem tool_faithful_injective [Functor.Faithful T] {A B : O} :'
'    Function.Injective (T.map : Quiver.Hom A B -> Quiver.Hom (T.obj A) (T.obj B)) :='
'  fun a b hab => T.map_injective hab'
''
'theorem no_agency_merging [Functor.Faithful T] {A B : O} {f g : Quiver.Hom A B}'
'    (h : T.map f = T.map g) : f = g :='
'  T.map_injective h'
''
'theorem empty_propagation [Functor.Full T] {A B : O} (h : Quiver.Hom A B -> False) :'
'    Quiver.Hom (T.obj A) (T.obj B) -> False := by'
'  intro f'
'  let g : Quiver.Hom A B := (T.map_surjective f).choose'
'  exact h g'
''
'theorem hom_equiv [Functor.Full T] [Functor.Faithful T] {A B : O} :'
'    Nonempty (Equiv (Quiver.Hom A B) (Quiver.Hom (T.obj A) (T.obj B))) :='
'  Nonempty.intro (Equiv.ofBijective T.map'
'    (And.intro T.map_injective T.map_surjective))'
''
'theorem no_autonomous_endomorphism [Functor.Full T] [Functor.Faithful T] {A : O}'
'    (h : forall (f : Quiver.Hom A A), f = CategoryStruct.id A)'
'    (g : Quiver.Hom (T.obj A) (T.obj A)) :'
'    g = CategoryStruct.id (T.obj A) := by'
'  let f : Quiver.Hom A A := (T.map_surjective g).choose'
'  have hf : T.map f = g := (T.map_surjective g).choose_spec'
'  rw [<- hf, h f, T.map_id]'
''
'end DIP.AxiomI'
)
